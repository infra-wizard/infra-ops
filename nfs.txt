apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: azurefile-nfs-encrypted
provisioner: file.csi.azure.com
parameters:
  protocol: nfs
  encryptInTransit: "true"
  resourceGroup: myResourceGroup      # RG above
  storageAccount: mynfsaccount12345   # $SA above
mountOptions:
  - nconnect=4
reclaimPolicy: Retain
volumeBindingMode: Immediate
allowVolumeExpansion: true


----

apiVersion: v1
kind: PersistentVolume
metadata:
  name: pv-nfs-encrypted
spec:
  capacity:
    storage: 100Gi
  accessModes:
    - ReadWriteMany
  persistentVolumeReclaimPolicy: Retain
  mountOptions:
    - nconnect=4
  csi:
    driver: file.csi.azure.com
    # volumeHandle must be unique per volume; convention: {resource-group}#{storage-account}#{file-share}
    volumeHandle: myResourceGroup#mynfsaccount12345#mynfsshare
    volumeAttributes:
      protocol: nfs
      encryptInTransit: "true"
      resourceGroup: myResourceGroup
      storageAccount: mynfsaccount12345
      shareName: mynfsshare


-----
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: pvc-nfs-encrypted
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: ""
  volumeName: pv-nfs-encrypted
  resources:
    requests:
      storage: 100Gi
-----

apiVersion: v1

  name: nfs-encrypted-pvc
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: azurefile-nfs-encrypted
  resources:
